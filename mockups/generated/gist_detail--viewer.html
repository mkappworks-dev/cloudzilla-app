Retry with jitter in Go

Aada-silva · May 13, 2025 · Public

13 lines · 268 B
package retry

import (
	"math/rand"
	"time"
)

// Backoff returns the wait before attempt n, with up to 50% jitter.
func Backoff(n int) time.Duration {
	base := time.Duration(1<<n) * 100 * time.Millisecond
	return base/2 + time.Duration(rand.Int63n(int64(base/2)))
}
